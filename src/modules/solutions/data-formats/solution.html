CSV-парсер с поддержкой кавычек и экранирования (regex-вариант) 
import * as fs from 'node:fs';
import * as readline from 'node:readline';

export type CsvRow = string[];


/**
  * Важно по результатам бенчмарка:
  * regex-вариант универсальнее, но заметно медленнее.
  * Упрощённый вариант:
  * return line.split(separator)
  * на текущем наборе данных работает сильно быстрее.
*/
export function splitCsvLine(line: string, separator = ','): CsvRow {
  // Экранируем спецсимволы RegExp в разделителе (например |, ., ?, +),
  // чтобы использовать separator как буквальный символ в шаблоне.
  const sep = separator.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

  return line
    // 1) Делим только по НЕэкранированному разделителю:
    //    (?<!\\)${sep}
    // 2) И только если разделитель вне quoted-поля:
    //    (?=(?:[^"]*"[^"]*")*[^"]*$)
    .split(new RegExp(`(?<!\\\\)${sep}(?=(?:[^"]*"[^"]*")*[^"]*$)`))
    .map((field) =>
      field
        // Снять внешние кавычки: "abc" -> abc
        .replace(/^"(.*)"$/, '$1')
        // Двойные кавычки внутри quoted-поля: "" -> "
        .replace(/""/g, '"')
        // Вернуть экранированный разделитель: \, -> ,
        .replace(new RegExp(`\\\\${sep}`, 'g'), separator)
    );
}

export function parseCSV(
  file: string,
  separator = ',',
  cb: (err: Error | null, data: CsvRow[]) => void
): void {
  const rows: CsvRow[] = [];
  const rl = readline.createInterface({
    input: fs.createReadStream(file),
    crlfDelay: Infinity
  });

  rl.on('line', (line) => {
    rows.push(splitCsvLine(line, separator));
  });

  rl.once('close', () => cb(null, rows));
  rl.once('error', (err) => cb(err, []));
}
